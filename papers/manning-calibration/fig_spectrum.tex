% The spectrum figure: how many parameters the observable supports, at the
% two prior widths the paper uses.
%
% Form: a sorted dot-and-stem plot on a log axis, not a bar chart. The data's
% job is to show WHERE a threshold falls in a sorted magnitude list, and the
% reference line at lambda = 1 is the whole point -- above it the data
% determines the direction, below it the prior does. A log axis is required
% because the values span four decades; a linear one would collapse most of
% the fifteen onto the floor and hide the cliff, which is the finding.
%
% Two series: the absolute prior sigma_n = 0.015 (filled blue) and the uniform
% fractional prior sigma_alpha = 0.30 (open orange). Identity is carried by
% marker fill as well as hue, so the figure survives greyscale printing.
%
% Sources: logs/o61/o61_spectrum_sigma_n0.015.txt and
% logs/o61/o61_spectrum_sigma_alpha0.30.txt (o58_gauss_newton.py on the
% sixteen o61 production peak dumps, sigma_obs 0.15).
\begin{tikzpicture}
\begin{axis}[
  width=0.88\linewidth, height=5.8cm,
  xlabel={eigendirection, sorted},
  ylabel={$\lambda$ \ (data vs.\ prior information)},
  xmin=0.3, xmax=15.7,
  ymin=3e-5, ymax=40,
  ymode=log,
  xtick={1,3,5,7,9,11,13,15},
  tick align=outside, tick pos=left,
  axis line style={gray!55},
  tick label style={font=\footnotesize, color=black!65},
  label style={font=\small},
  grid=major, grid style={gray!18, line width=0.3pt},
  legend style={font=\scriptsize, at={(0.98,0.98)}, anchor=north east,
                draw=gray!40, fill=white, fill opacity=0.93, text opacity=1,
                row sep=0.5pt, inner sep=3pt},
  clip=false,
]

% the lambda = 1 threshold: data information equals prior information
\addplot[gray!65, line width=0.8pt, dashed, forget plot]
  coordinates {(0.3,1) (15.7,1)};
\node[font=\scriptsize, color=black!60, anchor=north east]
  at (axis cs:15.6,0.85) {$\lambda = 1$: data $=$ prior};

% stems, absolute prior
\addplot[ycomb, gray!45, line width=0.7pt, forget plot]
  coordinates {(1,2.779) (2,0.6823) (3,0.6127) (4,0.324) (5,0.2255)
               (6,0.09968) (7,0.06136) (8,0.03434) (9,0.02256) (10,0.01849)
               (11,0.01133) (12,0.007279) (13,0.003104) (14,0.002383)
               (15,0.0003867)};

% absolute prior sigma_n = 0.015: one supported direction
\addplot[only marks, mark=*, mark size=2.3pt,
         color={rgb,255:red,42;green,120;blue,214},
         mark options={fill={rgb,255:red,42;green,120;blue,214}}]
  coordinates {(1,2.779) (2,0.6823) (3,0.6127) (4,0.324) (5,0.2255)
               (6,0.09968) (7,0.06136) (8,0.03434) (9,0.02256) (10,0.01849)
               (11,0.01133) (12,0.007279) (13,0.003104) (14,0.002383)
               (15,0.0003867)};
\addlegendentry{absolute prior, $\sigma_n = 0.015$: one above the line}

% uniform fractional prior sigma_alpha = 0.30: three supported directions
\addplot[only marks, mark=o, mark size=2.6pt, line width=0.9pt,
         color={rgb,255:red,235;green,104;blue,52}]
  coordinates {(1,12.73) (2,2.985) (3,1.128) (4,0.8529) (5,0.2548)
               (6,0.2386) (7,0.1577) (8,0.0876) (9,0.05021) (10,0.02444)
               (11,0.01454) (12,0.01121) (13,0.006023) (14,0.004086)
               (15,0.0005972)};
\addlegendentry{fractional prior, $\pm 30\%$: three above the line}

\node[font=\scriptsize, color=black!60, anchor=west, align=left]
  at (axis cs:1.6,0.0016) {the prior answers for the rest};
\end{axis}
\end{tikzpicture}
